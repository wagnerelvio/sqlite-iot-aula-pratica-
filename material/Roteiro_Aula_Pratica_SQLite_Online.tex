\section{Apresentação}
Este roteiro tem como base o documento \emph{Guia Prático de SQL para SQLite: Sistema de Monitoramento IoT}, disponibilizado para a aula. O exemplo utiliza uma base com dispositivos e leituras de sensores, explorando DDL, DML, DQL, junções, agregações, expressões de tabela comuns (CTEs) e funções de janela. Foram acrescentados exercícios de atualização, exclusão e controle de transações como ampliação didática.

\section{Objetivos de aprendizagem}
\begin{enumerate}
\item Criar tabelas e identificar tipos de dados, chaves primárias e estrangeiras.
\item Inserir, consultar, atualizar e excluir registros com comandos SQL.
\item Inspecionar a estrutura das tabelas no SQLite Online.
\item Aplicar filtros, ordenação, junções e funções de agregação.
\item Explorar CTEs e funções de janela em dados históricos.
\item Interpretar dados simulados de monitoramento IoT.
\end{enumerate}

\section{Modelo de dados e preparação}
A tabela \texttt{dispositivos} armazena o cadastro dos equipamentos (\texttt{id}, \texttt{nome}, \texttt{localizacao}, \texttt{status}). A tabela \texttt{leituras} registra cada medição (\texttt{id}, \texttt{dispositivo\_id}, \texttt{temperatura}, \texttt{umidade}, \texttt{timestamp}). A relação é de \textbf{um para muitos}: um dispositivo pode possuir diversas leituras.

\begin{enumerate}
\item Acesse \url{https://sqliteonline.com/}.
\item Confirme que o mecanismo selecionado é \textbf{SQLite}.
\item Utilize um banco de dados de teste e localize o editor SQL e o botão de execução (\emph{Run}).
\item Execute cada bloco na ordem indicada. Para evitar erros de tabela já existente, não repita os comandos \texttt{CREATE TABLE} depois da primeira execução.
\end{enumerate}

\atividade{1}{Criação das tabelas (DDL)}
\textbf{Objetivo:} utilizar \texttt{CREATE TABLE}, tipos de dados e restrições de integridade.

\subsection*{Passo 1: criar a tabela de dispositivos}
\begin{lstlisting}
CREATE TABLE dispositivos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    localizacao TEXT NOT NULL,
    status TEXT CHECK (
        status IN ('ATIVO', 'MANUTENCAO', 'INATIVO')
    ) DEFAULT 'ATIVO'
);
\end{lstlisting}

\subsection*{Passo 2: criar a tabela de leituras}
\begin{lstlisting}
CREATE TABLE leituras (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    dispositivo_id INTEGER,
    temperatura REAL,
    umidade REAL,
    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (dispositivo_id)
        REFERENCES dispositivos(id)
);
\end{lstlisting}

\subsection*{Passo 3: inspecionar a estrutura e o relacionamento}
\begin{lstlisting}
PRAGMA table_info('dispositivos');
PRAGMA table_info('leituras');
PRAGMA foreign_key_list('leituras');
\end{lstlisting}
\begin{lstlisting}
SELECT name, sql
FROM sqlite_master
WHERE type = 'table'
  AND name IN ('dispositivos', 'leituras');
\end{lstlisting}
\questoes{Qual a finalidade da chave primária? Por que \texttt{leituras} tem uma chave estrangeira? O que a restrição \texttt{CHECK} permite?}

\atividade{3}{Consultas básicas (DQL)}
\textbf{Objetivo:} utilizar \texttt{SELECT}, \texttt{WHERE} e \texttt{ORDER BY}.

\subsection*{Consulta A: dispositivos ativos}
\begin{lstlisting}
SELECT id, nome, localizacao
FROM dispositivos
WHERE status = 'ATIVO';
\end{lstlisting}
\noindent\textbf{Resultado esperado:} 2 dispositivos.\par

\subsection*{Consulta B: leituras acima de 25 graus Celsius}
\begin{lstlisting}
SELECT *
FROM leituras
WHERE temperatura > 25;
\end{lstlisting}
\noindent\textbf{Resultado esperado:} 3 leituras.\par

\subsection*{Consulta C: temperaturas em ordem decrescente}
\begin{lstlisting}
SELECT dispositivo_id, temperatura, timestamp
FROM leituras
ORDER BY temperatura DESC;
\end{lstlisting}
\questoes{Qual é a diferença entre \texttt{WHERE} e \texttt{ORDER BY}? Identifique a menor e a maior temperatura registrada.}

\atividade{4}{Relacionamento entre tabelas (JOIN)}
\textbf{Objetivo:} combinar cadastro de dispositivos com suas medições.
\begin{lstlisting}
SELECT
    d.nome AS dispositivo,
    d.localizacao,
    l.temperatura,
    l.umidade,
    l.timestamp
FROM dispositivos d
JOIN leituras l
    ON d.id = l.dispositivo_id
ORDER BY l.timestamp DESC;
\end{lstlisting}
\questoes{Por que \texttt{JOIN} é necessário? O que ocorre se não houver um dispositivo correspondente a uma leitura em um \texttt{INNER JOIN}?}

\atividade{5}{Funções de agregação}
\textbf{Objetivo:} calcular média, máximo, mínimo e quantidade de medições.
\begin{lstlisting}
SELECT
    d.nome,
    ROUND(AVG(l.temperatura), 2) AS temp_media,
    MAX(l.temperatura) AS temp_maxima,
    MIN(l.temperatura) AS temp_minima,
    COUNT(l.id) AS total_leituras
FROM dispositivos d
LEFT JOIN leituras l
    ON d.id = l.dispositivo_id
GROUP BY d.id, d.nome;
\end{lstlisting}

\atividade{6}{Atualização e exclusão de registros (DML)}
\textbf{Objetivo:} utilizar \texttt{UPDATE} e \texttt{DELETE} com condições.

\subsection*{Passo 1: atualizar o status}
\begin{lstlisting}
UPDATE dispositivos
SET status = 'ATIVO'
WHERE id = 3;

SELECT * FROM dispositivos;
\end{lstlisting}

\subsection*{Passo 2: inserir e excluir uma leitura de teste}
\begin{lstlisting}
INSERT INTO leituras
(dispositivo_id, temperatura, umidade)
VALUES (1, 35.0, 50.0);

SELECT * FROM leituras
WHERE temperatura = 35.0;
\end{lstlisting}
\begin{lstlisting}
DELETE FROM leituras
WHERE dispositivo_id = 1
  AND temperatura = 35.0
  AND umidade = 50.0;

SELECT * FROM leituras;
\end{lstlisting}
\noindent\textbf{Verificação:} após excluir a leitura de teste, a tabela deve voltar a conter 7 leituras.\par
\questoes{Qual o papel de \texttt{WHERE} em \texttt{UPDATE} e \texttt{DELETE}? O que pode acontecer sem essa cláusula?}

\atividade{7}{Consultas avançadas: CTEs e funções de janela}
\textbf{Objetivo:} analisar registros históricos de cada equipamento.

\subsection*{Consulta A: última medição de cada dispositivo}
\begin{lstlisting}
WITH leituras_rankeadas AS (
    SELECT
        d.nome AS dispositivo,
        d.localizacao,
        l.temperatura,
        l.umidade,
        l.timestamp,
        ROW_NUMBER() OVER (
            PARTITION BY l.dispositivo_id
            ORDER BY l.timestamp DESC, l.id DESC
        ) AS rn
    FROM dispositivos d
    JOIN leituras l
        ON d.id = l.dispositivo_id
)
SELECT
    dispositivo, localizacao,
    temperatura, umidade, timestamp
FROM leituras_rankeadas
WHERE rn = 1;
\end{lstlisting}

\subsection*{Consulta B: variação entre temperaturas consecutivas}
\begin{lstlisting}
SELECT
    d.nome AS dispositivo,
    l.timestamp,
    l.temperatura AS temp_atual,
    LAG(l.temperatura, 1) OVER (
        PARTITION BY l.dispositivo_id
        ORDER BY l.timestamp, l.id
    ) AS temp_anterior,
    ROUND(
        l.temperatura -
        LAG(l.temperatura, 1) OVER (
            PARTITION BY l.dispositivo_id
            ORDER BY l.timestamp, l.id
        ), 2
    ) AS variacao_temperatura
FROM leituras l
JOIN dispositivos d
    ON l.dispositivo_id = d.id;
\end{lstlisting}

\subsection*{Consulta C: temperatura média superior a 22 graus Celsius}
\begin{lstlisting}
WITH medias_dispositivos AS (
    SELECT
        d.id,
        d.nome AS dispositivo,
        AVG(l.temperatura) AS media_temp
    FROM dispositivos d
    JOIN leituras l
        ON d.id = l.dispositivo_id
    GROUP BY d.id, d.nome
)
SELECT
    dispositivo,
    ROUND(media_temp, 2) AS media_temperatura
FROM medias_dispositivos
WHERE media_temp > 22.0;
\end{lstlisting}
\questoes{O que é uma CTE? Como diferem \texttt{ROW\_NUMBER()} e \texttt{LAG()}? Que utilidade teriam em um sistema de sensores?}

\atividade{8}{Controle de transações (TCL)}
\textbf{Objetivo:} comparar \texttt{ROLLBACK} e \texttt{COMMIT}.

\subsection*{Passo 1: alteração desfeita}
\begin{lstlisting}
BEGIN TRANSACTION;

UPDATE dispositivos
SET status = 'INATIVO'
WHERE id = 1;

SELECT * FROM dispositivos WHERE id = 1;

ROLLBACK;

SELECT * FROM dispositivos WHERE id = 1;
\end{lstlisting}

\subsection*{Passo 2: alteração confirmada}
\begin{lstlisting}
BEGIN TRANSACTION;

UPDATE dispositivos
SET status = 'ATIVO'
WHERE id = 1;

COMMIT;
\end{lstlisting}
\questoes{Explique as diferenças entre \texttt{COMMIT} e \texttt{ROLLBACK}.}
